Rule: write the test first, watch it fail (RED), write the minimum code to pass (GREEN), then move on. No refactoring pass — code is production-sized to begin with.

Sprint ordering is chosen to maximise fast feedback: sprints 1–4 are pure Python with no I/O, run in <2 s each. Sprint 5 mocks I/O. Sprints 6–7 need Node.js + a running sidecar. Sprint 8 needs API keys.

Define LAYER8_TS_BLOCK as a module-level string constant. Implement apply_patch() as two private helpers — one per file — called in sequence:

def apply_patch(cr_dir: Path) -> None:
    _patch_compression_ts(cr_dir)   # Patch A — Layer 8 hook
    _patch_tsup_config(cr_dir)      # Patch B — add dist/compression/index.js entry

def _patch_compression_ts(cr_dir: Path) -> None:
    target = cr_dir / "src/compression/index.ts"
    src = target.read_text()
    ANCHOR = "  const compressedChars = calculateTotalChars(result);"
    if ANCHOR not in src:
        raise RuntimeError(f"Patch A anchor not found in {target}.")
    if "LEANCTX_SIDECAR_URL" in src:
        return  # idempotent
    target.write_text(src.replace(ANCHOR, LAYER8_TS_BLOCK + "\n" + ANCHOR))

def _patch_tsup_config(cr_dir: Path) -> None:
    target = cr_dir / "tsup.config.ts"
    src = target.read_text()
    OLD = 'entry: ["src/index.ts", "src/cli.ts"],'
    NEW = 'entry: ["src/index.ts", "src/cli.ts", "src/compression/index.ts"],'
    if OLD not in src:
        if "src/compression/index.ts" in src:
            return  # idempotent
        raise RuntimeError(f"Patch B anchor not found in {target}.")
    target.write_text(src.replace(OLD, NEW))

import subprocess, time, os
import httpx

def health_check(url: str) -> bool:
    try:
        r = httpx.get(f"{url}/health", timeout=2)
        return r.status_code == 200
    except Exception:
        return False

def wait_for_health(proc, url: str, timeout: float = 30) -> None:
    deadline = time.time() + timeout
    while not health_check(url):
        if time.time() > deadline:
            raise TimeoutError(f"{url} did not become healthy within {timeout}s")
        time.sleep(1)

def spawn_cr_shim(workdir: Path, port: int, extra_env: dict) -> subprocess.Popen:
    env = {**os.environ, "CR_SHIM_PORT": str(port), **extra_env}
    proc = subprocess.Popen(["node", str(workdir / "cr_shim.mjs")], env=env)
    wait_for_health(proc, f"http://127.0.0.1:{port}")
    return proc

def spawn_leanctx_sidecar(url: str, lingua_ratio: float) -> subprocess.Popen | None:
    if health_check(url):
        return None  # already running
    env = {**os.environ, "LEANCTX_SERVER_LINGUA_RATIO": str(lingua_ratio)}
    proc = subprocess.Popen(["leanctx-serve"], env=env)
    wait_for_health(proc, url)
    return proc

def setup_clawtypeor(workdir: Path, commit: str) -> None:
    workdir.mkdir(parents=True, exist_ok=True)
    if not (workdir / ".git").exists():
        subprocess.run(["git","clone",CR_REPO_URL,str(workdir)], check=True)
    subprocess.run(["git","checkout",commit], cwd=workdir, check=True)
    subprocess.run(["npm","ci"], cwd=workdir, check=True)
    apply_patch(workdir)
    subprocess.run(["npm","run","build"], cwd=workdir, check=True)

def compress_via_shim(messages: list[dict], shim_url: str) -> dict:
    r = httpx.post(f"{shim_url}", json={"messages": messages}, timeout=30)
    r.raise_for_status()
    return r.json()

def run_leg(leg: str, items, shim_url: str, lb_cfg) -> list[dict]:

        rec = {"leg": leg, "workload": item["workload"],
               "tokens_raw": tokens_raw, "tokens_compressed": tokens_compressed,
               "cr_stats": shim_result.get("stats"), "accuracy": None}
        if lb_cfg and item.get("question"):
            answer, in_tok = call_eval_llm(compressed, lb_cfg)
            rec["accuracy"] = (answer == item["gold"])
            rec["eval_input_tokens"] = in_tok
        records.append(rec)
    return records
